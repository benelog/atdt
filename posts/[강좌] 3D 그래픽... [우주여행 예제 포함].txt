   │                        ---_____  │

 h에 위치하는 평면(모니터)과 h'에 위치하는 평면(실제  나
타날 점이 위치하는 평면)사이에는 일정한 비례식이  성립하
게 됩니다. 삼각형의 닮은꼴과 같은경우 입니다.  이렇게 생
각하면 비례식을 이용해 h'위의 한점 p'의  h위에서의  좌표
p를 구할수 있습니다.

 h : h' = p : p'
 
 에서

      hp'
 p = ──
      h'

 라는 식을 도출할 수 있습니다. 그리고 이 p좌표가  모니터
의 표시 범위밖인지 아니면 모니터 내의 점인지를  확인하여
출력 여부를 결정하면 됩니다. 우주여행 소스에서는 범위 밖
이면 별이 사라지고 새로운 별을 생성하도록 되어 있습니다.
그리고 별이 점점 날아오다가 모니터에 이르는 순간 즉, Z좌
표가 h가 되는순간에 별을 없애야 하겠지만 갑자기 별이  사
라지면 이상하니까 계속 날아와서 시야를 벗어나는 순간  사
라지도록 하였습니다.

 DEFINT A-Z

 TYPE Object
    CoordinateX1  AS SINGLE
    CoordinateY1  AS SINGLE
    CoordinateX2  AS SINGLE
    CoordinateY2  AS SINGLE
    FrontDistance AS SINGLE
    SideDistanceX AS SINGLE
    SideDistanceY AS SINGLE
    ColorNumber   AS INTEGER
 END TYPE

 DIM SIN1   AS SINGLE, SIN2   AS SINGLE
 DIM COS1   AS SINGLE, COS2   AS SINGLE
 DIM ViewX  AS SINGLE, ViewY  AS SINGLE

 DIM X AS SINGLE, Y AS SINGLE, Z AS SINGLE

 InputNumber = VAL(COMMAND$)
 IF InputNumber <> 0 THEN MaxStarCount = InputNumber ELSE_
MaxStarCount = 20

 SIN1 = SIN(0): SIN2 = SIN(0)
 COS1 = COS(0): COS2 = COS(0)

 HandleX = 0: HandleY = 0
 VectorX = 0: VectorY = 0

 DIM Star(MaxStarCount) AS Object

 RANDOMIZE TIMER

 SCREEN 12

 WINDOW SCREEN (-40, 30)-(40, -30)


 DO

 IF VectorX = HandleX THEN
    SELECT CASE INT(RND * 400)
        CASE 100: HandleX = 400
        CASE 200: HandleX = 0
        CASE 300: HandleX = -400
        CASE ELSE
    END SELECT
 ELSE
    SELECT CASE HandleX
        CASE 400: VectorX = VectorX + 1
        CASE 0: VectorX = VectorX - SGN(VectorX)
        CASE -400: VectorX = VectorX - 1
    END SELECT
    SIN1 = SIN((3.14 / 180) * (VectorX / 1500))
    COS1 = COS((3.14 / 180) * (VectorX / 1500))
 END IF


 IF VectorY = HandleY THEN
    SELECT CASE INT(RND * 400)
        CASE 100: HandleY = 400
        CASE 200: HandleY = 0
        CASE 300: HandleY = -400
        CASE ELSE
    END SELECT
 ELSE
    SELECT CASE HandleY
        CASE 400: VectorY = VectorY + 1
        CASE 0: VectorY = VectorY - SGN(VectorY)
        CASE -400: VectorY = VectorY - 1
    END SELECT
    SIN2 = SIN((3.14 / 180) * (VectorY / 1500))
    COS2 = COS((3.14 / 180) * (VectorY / 1500))
 END IF

 FOR I = 1 TO MaxStarCount
   
    IF Star(I).ColorNumber = 0 THEN
       
        Star(I).ColorNumber = INT(RND * 7)
        Star(I).FrontDistance = 1600
        Star(I).SideDistanceX = INT(RND * 1600 - 800)
        Star(I).SideDistanceY = INT(RND * 1200 - 600)
       
        ViewX = 80 * Star(I).SideDistanceX / Star(I).FrontDistance
        ViewY = 80 * Star(I).SideDistanceY / Star(I).FrontDistance
       
        PSET (ViewX, ViewY), Star(I).ColorNumber

        Star(I).CoordinateX1 = ViewX: Star(I).CoordinateY1 = ViewY
        Star(I).CoordinateX2 = ViewX: Star(I).CoordinateY2 = ViewY
    ELSE
        LINE (Star(I).CoordinateX1, Star(I).CoordinateY1)_
-(Star(I).CoordinateX2, Star(I).CoordinateY2), 0
       
        Star(I).FrontDistance = Star(I).FrontDistance - 15

        IF Star(I).FrontDistance < 600 AND Star(I).ColorNumber_
< 8 THEN Star(I).ColorNumber = Star(I).ColorNumber + 8
        
        X = Star(I).SideDistanceX * COS2 + Star(I)._
SideDistanceY * SIN1 * SIN2 - Star(I).FrontDistance * COS1 * SIN2
        Y = Star(I).SideDistanceY * COS1 + Star(I)._
FrontDistance * SIN1
        Z = Star(I).SideDistanceX * SIN1 - Star(I)._
SideDistanceY * SIN1 * COS2 + Star(I).FrontDistance * COS1 * COS2

        Star(I).SideDistanceX = X
        Star(I).SideDistanceY = Y
        Star(I).FrontDistance = Z
       
        IF Star(I).FrontDistance > 0 THEN
            ViewX = 80 * Star(I).SideDistanceX / Star(I)._
FrontDistance
            ViewY = 80 * Star(I).SideDistanceY / Star(I)._
FrontDistance
            IF (ViewX >= -40 AND ViewX <= 40) AND (ViewY >= -30_
AND ViewY <= 30) THEN
                LINE (ViewX, ViewY)-(Star(I).CoordinateX1,_
Star(I).CoordinateY1), Star(I).ColorNumber
                Star(I).CoordinateX2 = Star(I).CoordinateX1
                Star(I).CoordinateY2 = Star(I).CoordinateY1
                Star(I).CoordinateX1 = ViewX
                Star(I).CoordinateY1 = ViewY
            ELSE
                Star(I).ColorNumber = 0
            END IF
        ELSE
            Star(I).ColorNumber = 0
        END IF
    END IF

 NEXT I

 LOOP WHILE INKEY$ = ""

 END
